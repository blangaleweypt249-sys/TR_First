// 按反馈补启动：掉线的电机重发初始化，5010 直接发重启命令，每台最多连续 3 次
static void Arm_InitCheck(void)
{
    static uint8_t cont;
    static MG5010E *j1[2] = {&left_arm.joint1, &right_arm.joint1};
    static RS00 *j2[2] = {&left_arm.joint2, &right_arm.joint2};
    static RS05 *j3[4] = {&left_arm.joint3, &left_arm.joint4, &right_arm.joint3, &right_arm.joint4};
    static uint8_t retry_5010[2];
    static uint8_t retry[6];

    if (++cont < 10U) // 10ms 一轮，100ms 查一次
        return;
    cont = 0U;

    for (uint8_t i = 0; i < 2U; i++) // 5010 收不到回执就重启，等于重新上电
    {
        if (HAL_GetTick() - j1[i]->last_tick < 1500U) // 重启一次要 1 秒，别连着刷
            retry_5010[i] = 0U;
        else if (retry_5010[i] < 3U)
        {
            (void)osSemaphoreAcquire(arm_5010_sem, 0U); // 重启期间不喂
            reboot_tick = HAL_GetTick();

            MG5010E_Reboot(j1[i]);
            retry_5010[i]++;
        }
    }

    if (reboot_tick != 0U && (HAL_GetTick() - reboot_tick) > 1000U)
    {
        reboot_tick = 0U;
        osSemaphoreRelease(arm_5010_sem);
    }

    for (uint8_t i = 0; i < 2U; i++) // RS00 掉线就补 主动上报+模式+使能
    {
        if (HAL_GetTick() - j2[i]->last_tick < 200U)
            retry[i] = 0U;
        else if (retry[i] < 3U)
        {
            RS00_SetActiveReporting(j2[i]);
            RS00_SetMode(j2[i], mit);
            RS00_Enable(j2[i]);
            retry[i]++;
        }
    }

    for (uint8_t i = 0; i < 4U; i++) // RS05 同上
    {
        if (HAL_GetTick() - j3[i]->last_tick < 200U)
            retry[i + 2U] = 0U;
        else if (retry[i + 2U] < 3U)
        {
            RS05_SetActiveReporting(j3[i]);
            RS05_SetMode(j3[i], mit);
            RS05_Enable(j3[i]);
            retry[i + 2U]++;
        }
    }
}

W5500 网口（Jetson 直连）
- W5500 SCK ↔ PB3；MISO ↔ PB4；MOSI ↔ PB5（SPI1，AF5）；NSS ↔ PA15（低有效，软件片选）。这四个脚由 `user/W5500.c` 管理，CubeMX 不要分配给其他外设。板载 Flash 继续使用 SPI6。
- 本次采用轮询与软件复位；W5500 RST 不可一直为低，INT 不需要接入固件。模块供电按实板丝印，与 STM32 共地。
- Jetson 有线 `192.168.1.50/24` ↔ W5500 `192.168.1.199/24`，UDP 5000；直连不设网关。
- 诊断帧固定 8 字节：`A5 01 <四字节序号，大端> <前六字节累加和低8位> 5A`。只回显来自 Jetson 有线 IP 的合法帧，不接管底盘或机械臂。
- 验收：把 `tests/w5500_probe.py` 放到 Jetson 执行 `python3 w5500_probe.py --count 100`。合法帧应原样回显，长度、首尾或校验错误的帧不应回显。
- 2026-11-23 台架：Jetson 有线 100 Mbps，合法帧 100/100 回显；往返平均 5.81 ms，最大 10.64 ms；非法帧未回显，随后合法帧仍可回显。
- 性能调试：将 `tests/w5500_probe.py` 和 `tests/w5500_benchmark.py` 放在 Jetson 同一目录；运行 `python3 w5500_benchmark.py --rates 80 100 150 250 80 100 --seconds 5`。
- 台架折中：保留 5 ms 轮询和控制任务优先级。RX 积压超过 16 个短帧的字节量时只重开 UDP，保留 IP 和物理链路。80 包/秒持续 20 秒 1600/1600 回显，P95 10.25 ms；100 包/秒短测两轮均 400/400。150 包/秒下主动丢弃积压，P95 从约 1278 ms 降到 102.19 ms，降回 80/100 包/秒后恢复无丢包。常态建议 80 包/秒，以上为 8 字节诊断帧，不代表实车控制时延。
- 若网线亮但没有 ARP，先从 SWD 看 `w5500_version_in`（正常 `0x04`），再看 `w5500_phy_in` 位 0 和 `w5500_socket_in`（UDP 为 `0x22`）。
